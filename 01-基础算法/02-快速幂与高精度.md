## 参考：快速幂模板（补充）

```cpp
typedef long long ll;
const ll MOD = 998244353;

ll qpow(ll a, ll b) {
    ll res = 1;
    while (b) {
        if (b & 1) res = res * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return res;
}
```

## 2 的快速幂函数 pow2

```cpp
long long pow2(int n) {
 long long res = 1; 
long long base = 2;
 while (n > 0) {
 if (n & 1) { // 等价 n%2==1
 res *= base;
 }
 base *= base;
 n >>= 1; // 等价 n /= 2 } return res; }
```

## 高精度乘法：res = res * a

```cpp
// 函数1：高精度乘法 res = res * a 
// 作用：快速幂遇到二进制位为1时，把答案乘上当前底数 
// ========================== 
void multiply1(){ 
int tmp[1010]={0}; // 临时数组，存乘法中间结果 // 核心：高精度乘法（只算前500位，因为只要最后500位） 
for(int i=0;i<500;i++){ // 遍历res的每一位 
for(int j=0;j<500;j++){ // 遍历a的每一位 
tmp[i+j] += res[i] * a[j];// 乘法结果加到对应位 
}
 } 
int t=0; // 进位 
// 处理进位，只保留前500位 
for(int i=0;i<500;i++){ 
tmp[i] += t; // 加上之前的进位 
res[i] = tmp[i] % 10; // 当前位只保留个位数 
t = tmp[i] / 10; // 新的进位 
}
 }
```
