## 矩阵乘法规则

- 矩阵 A × 矩阵 B = 新矩阵 C
- $C[i][j] = A$ 的第 $i$ 行 $\cdot$ $B$ 的第 $j$ 列

## 斐波那契递推式转化为矩阵乘法（矩阵加速）

核心步骤，也是矩阵加速的关键！把 f(n) = f(n-1) + f(n-2) 写成矩阵相乘的形式：

### 1. 先写基础等式

```
f(n)   = 1*f(n-1) + 1*f(n-2)
f(n-1) = 1*f(n-1) + 0*f(n-2)
```

> 第二个等式是「凑数」的，目的是把递推式转化为矩阵形式

### 2. 转化为矩阵乘法

$$\begin{bmatrix} f(n) \\ f(n-1) \end{bmatrix} = \begin{bmatrix} 1 & 1 \\ 1 & 0 \end{bmatrix} \times \begin{bmatrix} f(n-1) \\ f(n-2) \end{bmatrix}$$

### 3. 递推展开（关键！）

$$\begin{bmatrix} f(n) \\ f(n-1) \end{bmatrix} = \begin{bmatrix} 1 & 1 \\ 1 & 0 \end{bmatrix}^{n-2} \times \begin{bmatrix} f(2) \\ f(1) \end{bmatrix}$$

- 右边的 [f(2), f(1)] 是初始值（[1,1]）
- 中间的矩阵就是我们要算幂的「转移矩阵」（记为 A）
- 算 A^(n-2)，再乘以初始值矩阵，就能直接得到 f(n)！

## 参考：矩阵快速幂模板（补充）

```cpp
struct Mat {
    ll a[2][2];
    Mat() { memset(a, 0, sizeof a); }
    Mat operator*(const Mat &o) const {
        Mat r;
        for (int i = 0; i < 2; i++)
            for (int j = 0; j < 2; j++)
                for (int k = 0; k < 2; k++)
                    r.a[i][j] = (r.a[i][j] + a[i][k] * o.a[k][j]) % MOD;
        return r;
    }
};

Mat qpow(Mat base, ll n) {
    Mat res; res.a[0][0] = res.a[1][1] = 1; // 单位矩阵
    while (n) {
        if (n & 1) res = res * base;
        base = base * base;
        n >>= 1;
    }
    return res;
}

ll fib(ll n) {
    if (n <= 2) return 1;
    Mat base;                 // 转移矩阵 A = [[1,1],[1,0]]
    base.a[0][0] = 1; base.a[0][1] = 1;
    base.a[1][0] = 1; base.a[1][1] = 0;
    Mat A = qpow(base, n - 2);
    return (A.a[0][0] + A.a[0][1]) % MOD; // × [f(2), f(1)] = [1,1]
}
```

## 高斯消元模板

```cpp
const double eps = 1e-6; // 浮点数精度阈值：小于这个数视为0
int main() {
    int n; // 未知数个数 = 方程个数
    cin >> n;
    double a[105][105]; // 增广矩阵：n行，n+1列（前n列系数，最后1列常数项）
    // 输入增广矩阵
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n + 1; j++) {
            cin >> a[i][j];
        }
    }
    // ===================== 核心：高斯消元（保留你原有的循环结构）
    bool has_no_solution = false; // 标记是否无解
    bool has_multi_solution = false; // 标记是否无穷多解
    // 遍历每一列（每个未知数），i是当前处理的列号（也是目标主元行号）
    for (int i = 1; i <= n; i++) {
        // -------- 步骤1：选主元（找当前列绝对值最大的行，避免精度误差）
        int maxa = i; // 初始假设当前行是主元行
        for (int j = i + 1; j <= n; j++) { // 遍历当前行下方的所有行
            if (fabs(a[j][i]) > fabs(a[maxa][i])) { // 找绝对值最大的元素
                maxa = j; // 更新主元行号
            }
        }
        swap(a[maxa], a[i]); // 把主元行交换到当前第i行
        // -------- 步骤2：判断当前列是否全为0（原板子直接判无解，这里先标记，不直接退出）
        if (fabs(a[i][i]) < eps) {
            // 当前列全0，先不退出，继续处理下一列，最后统一判断解的类型
            continue;
        }
        // -------- 步骤3：主元归一化（把当前行的主元位置变成1）--------
        // 从后往前算：避免先改a[i][i]导致后续计算错误
        for (int j = n + 1; j >= i; j--) {
            a[i][j] /= a[i][i];
        }
        // -------- 步骤4：消去其他行的当前列（让其他行的第i列变成0）
        for (int j = 1; j <= n; j++) { // 遍历所有行
            if (j != i) { // 跳过主元行本身
                // 从后往前更新：避免前面的修改覆盖需要用的原始值
                for (int k = n + 1; k >= i; k--) {
                    a[j][k] -= a[i][k] * a[j][i];
                }
            }
        }
    }
    // ===================== 关键修改：统一判断解的类型（替换原板子的直接退出）=====================
    // 第一步：检查是否有「0=非0」的矛盾行（无解）
    for (int i = 1; i <= n; i++) {
        // 判断条件：当前行前n列系数全为0，但常数项≠0 → 0x1+0x2+...=非0 → 无解
        bool all_zero = true;
        for (int j = 1; j <= n; j++) {
            if (fabs(a[i][j]) > eps) { // 只要有一个系数≠0，就不是全0行
                all_zero = false;
                break;
            }
        }
        // 如果系数全0，但常数项≠0 → 无解
        if (all_zero && fabs(a[i][n + 1]) > eps) {
            has_no_solution = true;
            break;
        }
    }
    // 第二步：如果无解，直接输出
    if (has_no_solution) {
        cout << "No Solution" << endl;
        return 0;
    }
    // 第三步：检查是否有无穷多解（有效方程数 < 未知数个数）
    int valid_row = 0; // 统计有效方程数（主元非0的行）
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (fabs(a[i][j]) > eps) { // 只要有一个系数≠0，就是有效行
                valid_row++;
                break;
            }
        }
    }
    if (valid_row < n) { // 有效方程数 < 未知数个数 → 无穷多解
        has_multi_solution = true;
    }
    // ===================== 输出结果 =====================
    if (has_multi_solution) {
        cout << "Multiple Solutions" << endl;
    } else {
        // 唯一解：输出每个未知数的解（最后一列），保留2位小数
        for (int i = 1; i <= n; i++) {
            cout << fixed << setprecision(2) << a[i][n + 1] << '\n';
        }
    }
    return 0;
}
```
