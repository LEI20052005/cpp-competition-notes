# 数位 DP

> Windy 数：相邻两位数字差 ≥ 2。

```cpp
// dp[pos][pre][lead]
// pos:当前处理第pos位；pre：上一位填的数字；lead：是否处于前导零
typedef long long ll;
ll dp[15][10][2];
int num[15];

// pos:当前位，pre:上一位数字，lead:前导零，limit:是否受数字上界约束
ll dfs(int pos, int pre, bool lead, bool limit) {
    if (pos == 0) {
        // 递归到底，只有不是全前导零，才算一个合法数
        return lead ? 0 : 1;
    }
    // 记忆化：不受limit限制才可以复用dp
    if (!limit && dp[pos][pre][lead] != -1)
        return dp[pos][pre][lead];

    ll res = 0;
    int up = limit ? num[pos] : 9; // 当前位最大能填多少

    for (int i = 0; i <= up; ++i) {
        bool new_limit = limit && (i == up);
        bool new_lead = lead && (i == 0);

        if (new_lead) {
            // 依旧前导零，没有实际数字，pre无意义
            res += dfs(pos - 1, -2, true, new_limit);
        } else {
            if (lead) {
                // 刚刚结束前导零，第一位数字，没有前一位，直接递归
                res += dfs(pos - 1, i, false, new_limit);
            } else {
                // 普通情况：相邻数字差≥2才允许转移
                if (abs(i - pre) >= 2) {
                    res += dfs(pos - 1, i, false, new_limit);
                }
            }
        }
    }
    // 只有不受上限约束，存入dp记忆
    if (!limit)
        dp[pos][pre][lead] = res;
    return res;
}

// 计算 [1, x] windy数数量
ll count(ll x) {
    if (x == 0) return 0;
    memset(dp, -1, sizeof(dp)); // 记忆化初始-1，标记未计算
    int len = 0;
    while (x > 0) {
        num[++len] = x % 10;
        x /= 10;
    }
    return dfs(len, -2, true, true);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    ll a, b;
    cin >> a >> b;
    cout << count(b) - count(a - 1) << endl;
    return 0;
}
```
