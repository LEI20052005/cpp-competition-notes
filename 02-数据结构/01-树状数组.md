# 树状数组

```cpp
const int MAXN = 1000010;
int tree[MAXN]; // 树状数组本体，tree[i]存一段区间的和，区间长度lowbit(i)

// lowbit：取出x二进制最右侧的1，树状数组核心运算
inline int lowbit(int x) {
    return x & (-x);
}

// 单点更新：给下标pos的位置，加上val
// @param pos 数组下标，从1开始
// @param val 要增加的值，可以是1，也可以是其他数
void update(int pos, int val) {
    // 不断向上找父节点，把所有包含pos的节点全部更新
    while (pos < MAXN) {
        tree[pos] += val;
        pos += lowbit(pos); // 加上lowbit，跳到上层父节点
    }
}

// 查询前缀和：求 [1 ~ pos] 的总和
// @param pos 查询的右端点，下标从1开始
// @return 返回1~pos区间总和，注意大数据建议用long long，防止溢出
long long query(int pos) {
    long long res = 0;
    // 不断向前拆分区间，累加各个小段
    while (pos > 0) {
        res += tree[pos];
        pos -= lowbit(pos); // 减去lowbit，往前跳到下一个片段
    }
    return res;
}

// 求区间 [l, r] 的和，直接调用上面两个函数即可，不用再写循环
// sum(l,r) = query(r) - query(l-1);

int main() {
    // 原始数组 a[1]=2, a[2]=1, a[3]=4, a[4]=5, a[5]=3
    // 逐个单点插入树状数组
    update(1, 2);
    update(2, 1);
    update(3, 4);
    update(4, 5);
    update(5, 3);

    cout << query(3) << endl;        // 1~3总和：2+1+4 = 7
    cout << query(5) - query(2) << endl; // [3,5]总和：4+5+3 = 12
    return 0;
}
```
