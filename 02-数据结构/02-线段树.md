# 线段树

> 完整模板见下方（区间和 + 区间最大值，支持乘/加懒标记，先乘后加）。

## 线段树模板（区间和+最大值，乘加懒标记）

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5 + 10;
// 节点：区间和、区间最大值
struct Node {
ll sum;
ll maxv;
} tree[4 * N];
// 懒标记：乘法、加法，优先级【先乘后加】
struct Lazy {
ll mul;
ll add;
} lazy[4 * N];
ll a[N];
// 重置懒标记
void clearLazy(int node) {
lazy[node].mul = 1;
lazy[node].add = 0;
}
// 合并左右子节点
Node merge(Node L, Node R) {
Node res;
res.sum = L.sum + R.sum;
res.maxv = max(L.maxv, R.maxv);
return res;
}
// 懒标记下传
void pushdown(int node, int l, int r) {
if (l == r) return;
int mid = (l + r) / 2;
int ls = node * 2, rs = node * 2 + 1;
// 更新左孩子
tree[ls].sum = tree[ls].sum * lazy[node].mul + lazy[node].add * (mid - l + 1);
tree[ls].maxv = tree[ls].maxv * lazy[node].mul + lazy[node].add;
lazy[ls].mul *= lazy[node].mul;
lazy[ls].add = lazy[ls].add * lazy[node].mul + lazy[node].add;
// 更新右孩子
tree[rs].sum = tree[rs].sum * lazy[node].mul + lazy[node].add * (r - mid);
tree[rs].maxv = tree[rs].maxv * lazy[node].mul + lazy[node].add;
lazy[rs].mul *= lazy[node].mul;
lazy[rs].add = lazy[rs].add * lazy[node].mul + lazy[node].add;
clearLazy(node);
}
// 构建线段树
void build(int node, int l, int r) {
clearLazy(node);
if (l == r) {
tree[node].sum = a[l];
tree[node].maxv = a[l];
return;
}
int mid = (l + r) / 2;
build(node*2, l, mid);
build(node*2+1, mid+1, r);
tree[node] = merge(tree[node*2], tree[node*2+1]);
}
// 单点修改：pos位置赋值val
void updatePoint(int node, int l, int r, int pos, ll val) {
if (l == r) {
tree[node].sum = val;
tree[node].maxv = val;
return;
}
pushdown(node, l, r);
int mid = (l + r) / 2;
if (pos <= mid) updatePoint(node*2, l, mid, pos, val);
else updatePoint(node*2+1, mid+1, r, pos, val);
tree[node] = merge(tree[node*2], tree[node*2+1]);
}
// 区间 [ll,rr] 加 k
void updateAdd(int node, int l, int r, int ll, int rr, ll k) {
if (r < ll || l > rr) return;
if (ll <= l && r <= rr) {
tree[node].sum += k * (r - l + 1);
tree[node].maxv += k;
lazy[node].add += k;
return;
}
pushdown(node, l, r);
int mid = (l + r) / 2;
updateAdd(node*2, l, mid, ll, rr, k);
updateAdd(node*2+1, mid+1, r, ll, rr, k);
tree[node] = merge(tree[node*2], tree[node*2+1]);
}
// 区间 [ll,rr] 乘 k
void updateMul(int node, int l, int r, int ll, int rr, ll k) {
if (r < ll || l > rr) return;
if (ll <= l && r <= rr) {
tree[node].sum *= k;
tree[node].maxv *= k;
lazy[node].mul *= k;
lazy[node].add *= k;
return;
}
pushdown(node, l, r);
int mid = (l + r) / 2;
updateMul(node*2, l, mid, ll, rr, k);
updateMul(node*2+1, mid+1, r, ll, rr, k);
tree[node] = merge(tree[node*2], tree[node*2+1]);
}
// 区间查询，返回结构体
Node querySeg(int node, int l, int r, int ll, int rr) {
if (r < ll || l > rr) return {0, (ll)-1e18};
if (ll <= l && r <= rr) return tree[node];
pushdown(node, l, r);
int mid = (l + r) / 2;
Node L = querySeg(node*2, l, mid, ll, rr);
Node R = querySeg(node*2+1, mid+1, r, ll, rr);
return merge(L, R);
}
int main() {
ios::sync_with_stdio(false); cin.tie(0);
int n;
cin >> n;
for (int i = 1; i <= n; i++) cin >> a[i];
build(1, 1, n);
/*使用示例
Node res = querySeg(1,1,n,2,5);
ll sum_val = res.sum;
ll max_val = res.maxv;
*/
return 0;
}
扫描线 + 线段树求矩形面积并
先整体说明算法思路：
扫描线法把每个矩形拆成左右两条竖边（左边界标记 + 1、右边界标记 - 1），按 x 坐标从小到大扫描；用线段树维护当前扫描区间内被覆盖的 y 轴总长度，相邻两条扫描线的 x 差值 × 覆盖的 y 长度 = 这一横条的面积，累加得到总面积；y 坐标范围极大所以做离散化压缩。
// 最大矩形数量1e5，每条矩形2条竖边，所以边数组开2倍空间
const int N=100010;
// 扫描线结构体：存储一条竖边信息
struct Bian{
long long x;    // 竖边的x坐标
long long y1,y2;// 竖边上下y范围[y1,y2]
int mark;       // 标记：左边界+1（加入覆盖），右边界-1（移除覆盖）
}b[N*2]; // 最多2*N条竖边
// 线段树节点结构体
struct Node{
int cnt;    // 区间被覆盖的次数（标记懒标记）
long long len;// 区间内实际被覆盖的y轴总长度
}tree[N*2*4]; // 离散后y最多2N个，线段树4倍空间
vector<long long> y; // 存放所有y坐标，用于后续离散化
// 重载小于号：按x坐标升序排序竖边，实现从左到右扫描
bool operator < (const Bian &b1,const Bian &b2){
return b1.x<b2.x;
}
// 线段树向上更新核心函数：根据子节点/覆盖次数更新当前节点len
void pushup(int node,int l,int r){
// 如果当前区间整体被覆盖次数>0，直接取该离散区间真实y长度
if(tree[node].cnt>0)
tree[node].len=y[r+1]-y[l];
else{
// 覆盖次数为0：叶子节点无覆盖长度为0
if(l==r)
tree[node].len=0;
// 非叶子：左右子节点覆盖长度相加得到总长度
else
tree[node].len=tree[node*2].len+tree[node*2+1].len;
}
}
// 线段区间更新函数：给[ll,rr]离散区间增加mk(+1/-1)覆盖标记
void updata(int node,int l,int r,int ll,int rr,int mk){
// 当前节点区间[l,r]完全在目标区间[ll,rr]内，直接修改cnt并更新len
if(l>=ll&&r<=rr){
tree[node].cnt+=mk;
pushup(node,l,r);
return ;
}
// 二分拆分左右子树递归更新
int m=(l+r)/2;
if(ll<=m)
updata(node*2,l,m,ll,rr,mk);
if(rr>m)
    updata(node*2+1,m+1,r,ll,rr,mk);
// 子树更新完向上合并结果
pushup(node,l,r);
}
int main(){
// 关闭cin同步加速输入，适配1e5大数据量
ios::sync_with_stdio(false),cin.tie(0);
int n;
long long x11,x22,y11,y22;
cin>>n; // 读入矩形个数
// 第一步：拆分矩形为两条竖边，收集所有y坐标待离散化
for(int i=1;i<=n;i++){
cin>>x11>>y11>>x22>>y22;
y.push_back(y11); // 存入y集合
y.push_back(y22);
// 左竖边：x=x11，标记+1
b[i]={x11,y11,y22,1};
// 右竖边：x=x22，标记-1
b[i+n]={x22,y11,y22,-1};
}
// y坐标离散化三步：排序→去重，把超大范围y映射到小下标
sort(y.begin(),y.end());         // 排序
sort(b+1,b+1+2*n);               // 所有竖边按x从小到大排序
y.erase(unique(y.begin(),y.end()),y.end()); // 去重，unique把重复元素放末尾再删除
int num=y.size(); // 离散后y的总个数
long long ans=0;  // 最终总面积
// 第二步：从左向右逐条扫描竖边
for(int i=1;i<=n*2;i++){
// i>1时：前一条边到当前边的x差值 × 当前总覆盖y长度 = 条带面积，累加
if(i>1)
ans+=(b[i].x-b[i-1].x)*tree[1].len;
// 把当前边y1/y2映射成离散下标
int l=lower_bound(y.begin(),y.end(),b[i].y1)-y.begin();
int r=lower_bound(y.begin(),y.end(),b[i].y2)-y.begin()-1;
// 线段树更新对应y区间的覆盖次数
updata(1,0,num-2,l,r,b[i].mark);
}
// 输出总面积
cout<<ans;
return 0;
}
```
