# 线段树

> 笔记原文该主题的截图内容为 STL 与数论知识（已分别归入「07-STL 与技巧」「05-数学数论」），此处保留主题并补充标准模板。

## 参考：线段树标准模板（补充）

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 100010;

ll a[MAXN], sum[MAXN << 2], lazy[MAXN << 2];

void push_up(int rt) { sum[rt] = sum[rt << 1] + sum[rt << 1 | 1]; }

void push_down(int rt, int l, int r) {
    if (lazy[rt]) {
        int m = (l + r) >> 1;
        lazy[rt << 1] += lazy[rt];
        lazy[rt << 1 | 1] += lazy[rt];
        sum[rt << 1] += lazy[rt] * (m - l + 1);
        sum[rt << 1 | 1] += lazy[rt] * (r - m);
        lazy[rt] = 0;
    }
}

void build(int rt, int l, int r) {
    if (l == r) { sum[rt] = a[l]; return; }
    int m = (l + r) >> 1;
    build(rt << 1, l, m);
    build(rt << 1 | 1, m + 1, r);
    push_up(rt);
}

void update(int rt, int l, int r, int L, int R, ll val) {
    if (L <= l && r <= R) {
        sum[rt] += val * (r - l + 1);
        lazy[rt] += val;
        return;
    }
    push_down(rt, l, r);
    int m = (l + r) >> 1;
    if (L <= m) update(rt << 1, l, m, L, R, val);
    if (R > m) update(rt << 1 | 1, m + 1, r, L, R, val);
    push_up(rt);
}

ll query(int rt, int l, int r, int L, int R) {
    if (L <= l && r <= R) return sum[rt];
    push_down(rt, l, r);
    int m = (l + r) >> 1;
    ll res = 0;
    if (L <= m) res += query(rt << 1, l, m, L, R);
    if (R > m) res += query(rt << 1 | 1, m + 1, r, L, R);
    return res;
}
```
