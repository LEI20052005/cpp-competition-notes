# 基环树

## 内向基环树解题思路

| 求解目标 | 算法 |
|---|---|
| 最长链长度 L，答案 = L+2 | 拓扑排序剥树，统计链的最大深度 |
| 所有环长度的最小公倍数 LCM | 找环，求多数字的 LCM |

## 参考：内向基环树模板（拓扑排序找环 + 剥链，补充）

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 100010;

int n;
int to[N];        // 内向基环树：i -> to[i]
int in[N];        // 入度
int depth[N];     // 剥树时统计链的最大深度

void solve() {
    queue<int> q;
    for (int i = 1; i <= n; i++) in[to[i]]++;   // 统计入度
    for (int i = 1; i <= n; i++)
        if (in[i] == 0) q.push(i);              // 入度为0：链端点

    int L = 0;                                  // 最长链长度
    while (!q.empty()) {
        int u = q.front(); q.pop();
        L = max(L, depth[u]);                   // 链的最大深度
        int v = to[u];
        depth[v] = max(depth[v], depth[u] + 1);
        if (--in[v] == 0) q.push(v);
    }
    // 此时入度仍 > 0 的点全部在环上
    vector<int> cyc;
    for (int i = 1; i <= n; i++)
        if (in[i] > 0) cyc.push_back(i);

    // 找环长 + 求 LCM：遍历环
    long long lcm = 1;
    vector<bool> vis(n + 1, false);
    for (int s : cyc) {
        if (vis[s]) continue;
        int len = 0, u = s;
        do { vis[u] = true; u = to[u]; len++; } while (u != s);
        lcm = lcm / __gcd(lcm, 1LL * len) * len;  // 多数字 LCM
    }
    // 答案：ans = L + 2（最长链问题）或 lcm（环长问题），按题目取用
    cout << "最长链 L = " << L << "，答案 = " << L + 2 << endl;
    cout << "所有环长 LCM = " << lcm << endl;
}
```
